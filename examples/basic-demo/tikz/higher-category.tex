% The full SVG is deliberately a normal, formula-bearing TikZ figure.
% dvisvgm specials only mark formula glyph groups and record their exact
% rendered bounding boxes for the deterministic extraction pass.
\newcommand{\SNLFormula}[2]{%
  \special{dvisvgm:bbox new snl#1}%
  \special{dvisvgm:raw <g data-snl-formula="#1">}%
  #2%
  \special{dvisvgm:raw </g>}%
  \special{dvisvgm:raw <g data-snl-anchor="#1" data-snl-bbox="{?bbox snl#1}"/>}%
}
